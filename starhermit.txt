# StarHermit game manifest — see https://wiki.starhermit.com/docs/api/github-games.html
name=Poker
description=Play multiplayer no-limit Texas Hold'em, bluffing and betting with play-money chips to win the pot.
# The platform assigns this game a uid and uses it as the game's slug — the same
# value as its <uid>.starhermit.com address. Nothing here can set it: a name a
# developer could choose would be something two games could fight over.
owner=2712e04e-461b-4d23-81ae-e40b429128a8
# The authoritative rules referee: every online table is a room-bound session
# server.js runs (deck, betting, pots, showdown, timers, AI seats, stats, Elo,
# the match result and the leaderboard scores). The client never reports one.
server=server.js
# All sound is synthesized (native/audio.ts renders the old WebAudio cues into
# clips); there are no authored .opus clips by design — the asset audit skips
# the sfx requirements when it sees this marker.
sfx=synthesized
cover=cover.jpg

# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.23.0
engine.minor=20
engine.capabilities=platform,storage,clipboard,voice

# Keyboard controls (players rebind keys and pads in the runtime's Options).
# Every button also takes the pointer, touch, or Tab and Enter; the bet slider
# takes a drag or the arrow keys while focused.
control.fold=KeyF | Fold
control.check_call=KeyC | Check / call
control.bet_raise=KeyB | Bet / raise
control.all_in=KeyA | All-in
control.bet_more=ArrowUp+Equal | Raise the bet size
control.bet_less=ArrowDown+Minus | Lower the bet size
control.preset_min=Digit1 | Bet size: minimum
control.preset_half=Digit2 | Bet size: half the pot
control.preset_three=Digit3 | Bet size: three quarters of the pot
control.preset_pot=Digit4 | Bet size: the pot
control.preset_all=Digit5 | Bet size: all-in
control.chat=KeyT | Table chat
engine.group.fold=Actions
engine.group.bet_more=Bet size
engine.group.chat=Table

# Gamepad defaults for the same actions (the D-pad or left stick and South
# move and press the UI's focus; East goes back).
engine.gamepad.fold=LeftTrigger
engine.gamepad.check_call=West
engine.gamepad.bet_raise=North
engine.gamepad.all_in=RightTrigger
engine.gamepad.bet_more=RightShoulder
engine.gamepad.bet_less=LeftShoulder
engine.gamepad.chat=Select
